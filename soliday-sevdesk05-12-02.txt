// Keys live only in sessionStorage – cleared when tab closes, never in source
function getKeys() {
  return {
    anthropic: sessionStorage.getItem("sd_ant_key") || "",
    sevdesk:   sessionStorage.getItem("sd_sev_key") || ""
  };
}

function saveKeys() {
  const ant = document.getElementById("inputAnthropicKey").value.trim();
  const sev = document.getElementById("inputSevdeskKey").value.trim();
  if (!ant || !sev) { alert("Bitte beide Keys eingeben."); return; }
  sessionStorage.setItem("sd_ant_key", ant);
  sessionStorage.setItem("sd_sev_key", sev);
  document.getElementById("btnResetKeys").style.display = "";
  document.getElementById("stepsBar").classList.remove("hidden");
  showView("viewUpload");
  setStep("upload");
}

function resetKeys() {
  sessionStorage.removeItem("sd_ant_key");
  sessionStorage.removeItem("sd_sev_key");
  document.getElementById("btnResetKeys").style.display = "none";
  document.getElementById("stepsBar").classList.add("hidden");
  document.getElementById("inputAnthropicKey").value = "";
  document.getElementById("inputSevdeskKey").value = "";
  showView("viewSetup");
}

// PDF -> Claude (mit Haiku → Sonnet Fallback für große Angebote)
async function callClaude(base64, model) {
  const { anthropic } = getKeys();
  const resp = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": anthropic,
      "anthropic-version": "2023-06-01",
      "anthropic-dangerous-direct-browser-access": "true"
    },
    body: JSON.stringify({
      model: model,
      max_tokens: 8192,
      system: SYSTEM_PROMPT,
      messages: [{
        role: "user",
        content: [
          { type: "document", source: { type: "base64", media_type: "application/pdf", data: base64 } },
          { type: "text", text: "Extrahiere alle Positionen aus diesem Soliday-Angebot." }
        ]
      }]
    })
  });

  if (!resp.ok) {
    const err = await resp.json().catch(() => ({}));
    const errMsg = err.error?.message || JSON.stringify(err);
    // Context-Fehler als speziellen Typ markieren für Fallback-Erkennung
    if (resp.status === 400 && /context|length|too long|too large/i.test(errMsg)) {
      const e = new Error(errMsg);
      e.isContextError = true;
      throw e;
    }
    throw new Error(`Anthropic API Fehler ${resp.status}: ${errMsg}`);
  }

  return resp.json();
}

async function processFile(file) {
  showView("viewLoading");
  setStep("upload");
  document.getElementById("progressBar").style.width = "15%";
  document.getElementById("loadingSub").textContent = "PDF wird gelesen...";

  try {
    const base64 = await new Promise((res, rej) => {
      const r = new FileReader();
      r.onload = () => res(r.result.split(",")[1]);
      r.onerror = () => rej(new Error("Datei konnte nicht gelesen werden"));
      r.readAsDataURL(file);
    });

    document.getElementById("progressBar").style.width = "35%";
    document.getElementById("loadingSub").textContent = "Claude Haiku analysiert Positionen...";

    let data;
    try {
      data = await callClaude(base64, "claude-haiku-4-5-20251001");
    } catch(e) {
      if (e.isContextError) {
        // Angebot zu groß für Haiku → Fallback auf Sonnet
        document.getElementById("progressBar").style.width = "45%";
        document.getElementById("loadingSub").textContent = "Großes Angebot – wechsle zu Claude Sonnet...";
        data = await callClaude(base64, "claude-sonnet-4-6");
      } else {
        throw e;
      }
    }

    document.getElementById("progressBar").style.width = "75%";

    const text = (data.content || []).find(b => b.type === "text")?.text || "";
    if (!text) throw new Error("Leere Antwort von Claude");

    const clean = text.replace(/```json\n?|```\n?/g, "").trim();
    if (!clean.endsWith("}")) {
      throw new Error("Claude-Antwort wurde abgeschnitten (Angebot zu groß). Bitte erneut versuchen – Sonnet wird automatisch verwendet.");
    }
    extractedData = JSON.parse(clean);

    document.getElementById("progressBar").style.width = "100%";
    setTimeout(renderPreview, 300);
  } catch(e) {

// sevDesk API
async function sevGet(path) {
  const { sevdesk } = getKeys();
  const r = await fetch(`${SEVDESK_BASE}${path}`, { headers: { Authorization: sevdesk } });
  if (!r.ok) throw new Error(`sevDesk GET ${path} HTTP ${r.status}`);
  return r.json();
}

async function sevPost(path, body) {
  const { sevdesk } = getKeys();
  const r = await fetch(`${SEVDESK_BASE}${path}`, {
    method: "POST",
    headers: { Authorization: sevdesk, "Content-Type": "application/json" },
    body: JSON.stringify(body)
  });
  const responseText = await r.text();
  let responseJson;
  try { responseJson = JSON.parse(responseText); } catch { responseJson = responseText; }

  if (!r.ok) {
    const pretty = obj => JSON.stringify(obj, null, 2).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
    document.getElementById("errorMsg").innerHTML =
      `<strong style="color:#ff8888">HTTP ${r.status} – ${responseJson?.error?.message || "Bad Request"} (errorCode ${responseJson?.error?.errorCode || "?"})</strong>\n\n` +
      `<span style="color:#888">POST ${path}</span>\n\n` +
      `<strong style="color:#ccc">REQUEST BODY:</strong>\n${pretty(body)}\n\n` +
      `<strong style="color:#ccc">RESPONSE:</strong>\n${pretty(responseJson)}`;
    showView("viewError");
    setStep("error");
    throw new Error("__debug__");
  }
  return responseJson;
}

// Betreff und headText aus erkanntem System ableiten
function getBetreff(system, snap, kommission) {
  const s = (system || "").toUpperCase();
  let base;
  if (s === "SOLIDAY-C")   base = "Angebot - SOLIDAY C";
  else if (s === "SOLIDAY-CS")  base = "Angebot - SOLIDAY CS";
  else if (s === "SOLIDAY-X")   base = "Angebot - SOLIDAY X";
  else if (s === "SOLIDAY-XS")  base = "Angebot - SOLIDAY XS";
  else if (s === "SOLIDAY-M")   base = "Angebot - SOLIDAY M";
  else if (s === "SOLIDAY-MA")  base = "Angebot - SOLIDAY MA";
  else if (s === "SOLIDAY-ONE") base = "Angebot - SOLIDAY ONE";
  else if (s === "SOLIDAY-FLEX") base = "Angebot - SOLIDAY FLEX";
  else if (s === "SOLIDAY-SANDY") base = "Angebot - SOLIDAY SANDY";
  else if (s === "SONNENSEGEL-MAß" || s === "SONNENSEGEL-MAß-SNAP") {
    base = snap ? "Angebot - SOLIDAY Fix Segel mit SNAP System" : "Angebot - SOLIDAY Fix Segel";
  }
  else if (s === "RAFF-M-DRAHTSEIL")       base = "Angebot - RAFF M Drahtseil";
  else if (s === "RAFF-M-FÜHRUNGSSCHIENE") base = "Angebot - RAFF M Führungsschiene";
  else base = "Angebot";
  const kom = (kommission || "").trim();
  return kom ? `${base} - ${kom}` : base;
}

function getHeadText(system, snap) {
  const s = (system || "").toUpperCase();
  if (s === "SOLIDAY-SANDY")  return "Sonnensegel & Sandkastenabdeckung";
  if (s === "SONNENSEGEL-MAß" || s === "SONNENSEGEL-MAß-SNAP") {
    return snap ? "Festes Sonnensegel nach Maß mit komfortabler SNAP System Bedienung"
                : "Festes Sonnensegel nach Maß";
  }
  if (s === "SOLIDAY-FLEX")   return "Robustes aufrollbares Sonnensegel mit SNAP System";
  if (s === "SOLIDAY-M" || s === "SOLIDAY-MA") return "Manuell aufrollbares Sonnensegel";
  if (s === "SOLIDAY-ONE")    return "Manuell aufrollbares Premium Sonnensegel mit Ein-Punkt Bedienung";
  if (s === "SOLIDAY-X")      return "Semi-automatisch aufrollbares Sonnensegel";
  if (s === "SOLIDAY-XS")     return "Semi-automatisch aufrollbares Markisensegel";
  if (s === "SOLIDAY-C")      return "Elektrisch aufrollbares Sonnensegel";
  if (s === "SOLIDAY-CS")     return "Elektrisch aufrollbares Markisensegel";
  if (s === "RAFF-M-DRAHTSEIL")       return "Manuell raffbares Sonnensegel mit Drahtseil Aufhängung";
  if (s === "RAFF-M-FÜHRUNGSSCHIENE") return "Manuell raffbares Sonnensegel mit Führungsschiene";

}

async function startTransfer() {
  showView("viewSending");
  setStep("sending");
  const bar = document.getElementById("sendBar");
  const sub = document.getElementById("sendingSub");

  try {
    sub.textContent = "Benutzer wird geladen...";
    bar.style.width = "15%";
    const sevUserData = await sevGet("/SevUser?limit=1");
    const sevUser = (sevUserData.objects || [])[0];
    if (!sevUser) throw new Error("SevUser nicht gefunden – bitte API-Key pruefen.");

    sub.textContent = "Import-Kunde wird gesucht...";
    bar.style.width = "30%";
    const contacts = await sevGet("/Contact?limit=200&depth=0");
    const allContacts = contacts.objects || [];
    const customer = allContacts.find(c =>
      (c.name||"").toLowerCase().includes("import") ||
      (c.customerNumber||"").toLowerCase().includes("import")
    );
    if (!customer) {
      const sample = allContacts.slice(0,8).map(c => `"${c.name}" (${c.customerNumber})`).join(", ");
      throw new Error(`Import-Kunde nicht gefunden.\nVorhandene (erste 8): ${sample}`);
    }

    sub.textContent = "Einheiten werden geladen...";
    bar.style.width = "45%";
    const unitiesData = await sevGet("/Unity?limit=50");
    const unities = unitiesData.objects || [];
    const defaultUnity = unities.find(u => /stk|stueck|stück|pc|pcs/i.test(u.name))
                      || unities.find(u => String(u.id) === "1")
                      || unities[0];
    if (!defaultUnity) throw new Error("Keine Einheiten gefunden.");

    function findUnity(einheit) {
      const n = (einheit||"").toLowerCase().replace(/\./g,"").trim();
      if (["m2","m\u00b2","qm"].includes(n))
        return unities.find(u => /m.?2|qm/i.test(u.name)) || defaultUnity;
      if (["m","lm","lfm"].includes(n))
        return unities.find(u => /^(lm|lfm|m)$/i.test((u.name||"").trim())) || defaultUnity;
      if (["h","std"].includes(n))
        return unities.find(u => /std|stunde|hour/i.test(u.name)) || defaultUnity;
      return defaultUnity;
    }

    sub.textContent = "Positionen werden aufgebaut...";
    bar.style.width = "60%";

    const orderPosSave = extractedData.positionen.map((pos, i) => ({
      objectName: "OrderPos",
      mapAll: "true",
      part: null,
      quantity: Number(pos.menge),
      price: String(Number(pos.preis_einheit).toFixed(2)),
      name: String(pos.name),
      unity: { id: String(findUnity(pos.einheit).id), objectName: "Unity" },
      positionNumber: i,
      text: null,
      discount: 0,
      optional: false,
      taxRate: 19
    }));

    // Teuerungszuschlag: als Festbetrag-Aufschlag im discounts-Array
    // sevdesk API: discount=false → Aufschlag; percentage=false → Festbetrag (Netto)
    const teuerung = extractedData.teuerungszuschlag_netto;
    const hasTeuerung = teuerung && Number(teuerung) !== 0;

    sub.textContent = "Angebotsnummer wird von sevdesk geholt...";
    bar.style.width = "65%";
    const nextNumData = await sevGet("/Order/Factory/getNextOrderNumber?orderType=AN&useNextNumber=true");
    const orderNumber = nextNumData.objects || nextNumData.number || `AN-${Date.now()}`;

    const today = new Date();
    const pad = n => String(n).padStart(2, "0");
    const offerDate = `${pad(today.getDate())}.${pad(today.getMonth()+1)}.${today.getFullYear()}`;

    const addrParts = [
      customer.name,
      customer.street || null,
      (customer.zipCode || customer.zip) && customer.city
        ? `${customer.zipCode || customer.zip} ${customer.city}`
        : customer.city || null
    ].filter(Boolean);
    const address = addrParts.join("\n") || customer.name || "Import";

    sub.textContent = "Angebot wird in sevdesk angelegt...";
    bar.style.width = "80%";

    const payload = {
      order: {
        id: null,
        objectName: "Order",
        mapAll: true,
        orderNumber: orderNumber,
        contact: { id: String(customer.id), objectName: "Contact" },
        orderDate: offerDate,
        status: 100,
        header: getBetreff(extractedData.system, extractedData.snap, extractedData.kommission),
        headText: getHeadText(extractedData.system, extractedData.snap),
        footText: FOOT_TEXT,
        address: address,
        addressCountry: { id: "1", objectName: "StaticCountry" },
        version: 0,
        smallSettlement: 0,
        contactPerson: { id: String(sevUser.id), objectName: "SevUser" },
        taxRate: 19,
        discount: hasTeuerung ? 0 : 0,
        discountTime: hasTeuerung ? 1 : 0,
        taxSet: null,
        taxText: "Umsatzsteuer 19%",
        taxRule: { id: "1", objectName: "TaxRule" },
        orderType: "AN",
        currency: "EUR",
        showNet: 1,
        sendType: "VPR"
      },
      orderPosSave: orderPosSave,
      orderPosDelete: null,
      discountSave: hasTeuerung ? [{
        objectName: "Discounts",
        mapAll: true,
        discount: false,
        text: "Temporäre Anpassung aufgrund von Rohstoff- & Energiekosten",
        percentage: false,
        value: Number(Number(teuerung).toFixed(2))
      }] : null,
      discountDelete: null
    };

    const offerResp = await sevPost("/Order/Factory/saveOrder", payload);
    const offer = offerResp?.objects?.order || offerResp?.objects || offerResp;

    if (!offer?.id) {
      throw new Error("Keine Angebots-ID.\nAntwort: " + JSON.stringify(offerResp).substring(0,400));
    }

    bar.style.width = "100%";
    document.getElementById("doneLink").href = `https://my.sevdesk.de/#/ar/order/detail/id/${offer.id}`;
    setTimeout(() => { showView("viewDone"); setStep("done"); }, 300);

  } catch(e) {
    if (e.message !== "__debug__") showError(e.message);
  }
}
